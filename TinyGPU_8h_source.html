1// TinyFonts - optional integration with TinyGPU.
2//
3// TinyFonts itself does not depend on TinyGPU. Include this header only in
4// projects that use TinyGPU:
5//
6// #include <TinyGPU.h>
7// #include <TinyFonts/Integrations/TinyGPU.h>
8// #include <TinyFonts/Fonts/Lato16.h> // bundled, or generated by the converter
9//
10// tinyfonts::TinyGPUFont<RGB565> lato16(tinyfonts::Lato16);
11// framebuffer.setFont(lato16); // now TinyGPU drawing uses TinyFonts
12// framebuffer.drawText(8, 8, "Grüße", RGB565::fromRGB(255, 255, 0));
13//
14// TinyGPUFont implements tinygpu::IFont, so every TinyGPU surface (Surface,
15// FrameBuffer, sprites) can use it through setFont()/drawText(). Behaviour:
16// - Text is alpha-blended over what is already on the surface, unless
17// `opaque` is true, in which case the text box is filled with `background`
18// first.
19// - `scale` is an integer multiplier: each font pixel becomes a scale x scale
20// block. TinyFonts glyphs are sized at conversion time, so use a larger
21// font size for smooth large text.
22// - `spacing` and `lineSpacing` are accepted but ignored: TinyFonts uses each
23// glyph's own advance and the font's line gap.
24// - The surface clip rectangle (pushClipRect) is honoured.
25// - Fonts are always proportional; setProportional() has no effect.
26// - Multi-line text honours '\n'. Use drawWrapped() for word wrapping, since
27// IFont::drawText has no width parameter.
28//
29// Colours are handled through getRed/getGreen/getBlue and fromRGB, so the
30// template works with the TinyGPU colour types that provide those (RGB565 is
31// tested).
32

59template <typename RGB_T = tinygpu::RGB565>
60class TinyGPUDisplay : public FontDisplay {
61 public:
63 explicit TinyGPUDisplay(tinygpu::ISurface<RGB_T>& target) : target_(target) {}
64
66 void begin(int originX, int originY, RGB_T color, uint8_t scale) {
67 originX_ = originX;
68 originY_ = originY;
69 color_ = color;
70 scale_ = scale == 0 ? 1 : scale;
71 }
72
74 void drawPixel(int x, int y, uint8_t coverage) override {
75 if (coverage == 0) return;
76 for (int dy = 0; dy < scale_; ++dy) {
77 for (int dx = 0; dx < scale_; ++dx) {
78 const int px = originX_ + x * scale_ + dx;
79 const int py = originY_ + y * scale_ + dy;
80 if (px < 0 || py < 0 || px >= static_cast<int>(target_.width()) ||
81 py >= static_cast<int>(target_.height())) {
82 continue;
83 }
84 if (!target_.isClipVisible(px, py)) continue;
85 if (coverage == 255) {
86 target_.setPixel(px, py, color_);
87 } else {
88 target_.setPixel(px, py,
89 blend(target_.getPixel(px, py), color_, coverage));
90 }
91 }
92 }
93 }
94
95 // Linear blend: fg * a + bg * (1 - a), with a in 0..255.
97 static RGB_T blend(RGB_T bg, RGB_T fg, uint8_t a) {
98 return RGB_T::fromRGB(mix(bg.getRed(), fg.getRed(), a),
99 mix(bg.getGreen(), fg.getGreen(), a),
100 mix(bg.getBlue(), fg.getBlue(), a));
101 }
102
103 private:
104 static uint8_t mix(uint8_t bg, uint8_t fg, uint8_t a) {
105 return static_cast<uint8_t>((bg * (255 - a) + fg * a + 127) / 255);
106 }
107
108 tinygpu::ISurface<RGB_T>& target_;
109 int originX_ = 0;
110 int originY_ = 0;
111 RGB_T color_;
112 int scale_ = 1;
113};

195 // Shared drawing path. Returns the height drawn, in pixels.
196 size_t drawBlock(tinygpu::ISurface<RGB_T>& target, int16_t x, int16_t y,
197 const char* text, int maxWidth, RGB_T foreground,
198 RGB_T background, bool opaque, uint8_t scale) const {
199 if (text == nullptr) return 0;
200 const int s = scale == 0 ? 1 : scale;
201 const TextBox box = measure(text, maxWidth / s);
202
203 if (opaque) {
204 fillBox(target, x, y, box.width * s, box.height * s, background);
205 }
206
207 TinyGPUDisplay<RGB_T> display(target);
208 display.begin(x, y, foreground, static_cast<uint8_t>(s));
209 TextRenderer renderer(font_, display);
210 renderer.drawText(0, 0, text, maxWidth / s);
211 return static_cast<size_t>(box.height) * s;
212 }
213
214 // Fills a rectangle, clipped to the surface bounds.
215 static void fillBox(tinygpu::ISurface<RGB_T>& target, int x, int y, int w,
216 int h, RGB_T color) {
217 int x0 = x < 0 ? 0 : x;
218 int y0 = y < 0 ? 0 : y;
219 int x1 = x + w;
220 int y1 = y + h;
221 if (x1 > static_cast<int>(target.width())) x1 = static_cast<int>(target.width());
222 if (y1 > static_cast<int>(target.height())) y1 = static_cast<int>(target.height());
223 if (x1 <= x0 || y1 <= y0) return;
224 target.fillRect(x0, y0, x1 - x0, y1 - y0, color);
225 }

227 Font font_;
228};
