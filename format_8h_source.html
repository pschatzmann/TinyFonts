9// Binary layout of a TinyFonts blob. All multi-byte integers are little-endian.
10//
11// Header (24 bytes)
12// 0 'T' 'N' 'F' '1' magic
13// 4 u8 bitsPerPixel 1, 2, 4 or 8 (coverage depth of each pixel)
14// 5 u8 ascent pixels above the baseline
15// 6 u8 descent pixels below the baseline
16// 7 u8 lineGap extra pixels between lines
17// 8 u16 glyphCount
18// 10 u16 reserved
19// 12 u32 kerningCount 32-bit: large Unicode sets have many pairs
20// 16 u32 fallbackCodepoint used for characters that are not in the font
21// 20 u32 bitmapBase offset of the bitmap area from the blob start
22//
23// Glyph table: glyphCount entries of 16 bytes, sorted by codepoint
24// 0 u32 codepoint
25// 4 u32 bitmapOffset relative to bitmapBase
26// 8 u8 width bitmap width in pixels
27// 9 u8 height bitmap height in pixels
28// 10 i8 left x offset of the bitmap from the pen position
29// 11 i8 top distance from the baseline up to the top row
30// 12 u16 advance pen advance in pixels
31// 14 2 bytes reserved
32//
33// Kerning table: kerningCount entries of 10 bytes, sorted by (left, right)
34// 0 u32 left codepoint
35// 4 u32 right codepoint
36// 8 i16 adjustment in pixels (added after `left`, before `right`)
37//
38// Bitmap area: each glyph is stored row by row, most significant bit first,
39// with bitsPerPixel bits per pixel. Every row starts on a byte boundary.
40

89 // Raw little-endian accessors (offset is relative to the blob start).
91 uint8_t u8(size_t offset) const {
92#if TF_HAS_PGM
93 if (_inFlash) return pgm_read_byte(_data + offset);
94#endif
95 return _data[offset];
96 }
98 uint16_t u16(size_t offset) const {
99 return static_cast<uint16_t>(u8(offset) | (u8(offset + 1) << 8));
100 }
102 uint32_t u32(size_t offset) const {
103 return static_cast<uint32_t>(u16(offset)) |
104 (static_cast<uint32_t>(u16(offset + 2)) << 16);
105 }
107 int8_t i8(size_t offset) const { return static_cast<int8_t>(u8(offset)); }
109 int16_t i16(size_t offset) const { return static_cast<int16_t>(u16(offset)); }

135inline size_t glyphEntryOffset(uint16_t index) {
136 return kHeaderSize + static_cast<size_t>(index) * kGlyphEntrySize;
137}

140inline size_t kerningEntryOffset(const Font& font, size_t index) {
141 return kHeaderSize + static_cast<size_t>(font.glyphCount()) * kGlyphEntrySize +
142 static_cast<size_t>(index) * kKerningEntrySize;

146inline bool lookupGlyph(const Font& font, uint32_t codepoint, Glyph& out) {
147 size_t lo = 0;
148 size_t hi = font.glyphCount();
149 while (lo < hi) {
150 const size_t mid = lo + (hi - lo) / 2;
151 if (font.u32(glyphEntryOffset(static_cast<uint16_t>(mid))) < codepoint) {
152 lo = mid + 1;
153 } else {
154 hi = mid;
155 }
156 }
157 if (lo >= font.glyphCount()) return false;
158 const size_t o = glyphEntryOffset(static_cast<uint16_t>(lo));
159 if (font.u32(o) != codepoint) return false;
160 out.codepoint = codepoint;
161 out.bitmapOffset = font.u32(o + 4);
162 out.width = font.u8(o + 8);
163 out.height = font.u8(o + 9);
164 out.left = font.i8(o + 10);
165 out.top = font.i8(o + 11);
166 out.advance = font.u16(o + 12);
167 return true;
168}

171inline bool findGlyph(const Font& font, uint32_t codepoint, Glyph& out) {
172 if (!font.valid()) return false;
173 if (lookupGlyph(font, codepoint, out)) return true;
174 const uint32_t fallback = font.fallbackCodepoint();
175 return fallback != codepoint && lookupGlyph(font, fallback, out);
176}
177
179inline int kerning(const Font& font, uint32_t left, uint32_t right) {
180 if (!font.valid()) return 0;
181 size_t lo = 0;
182 size_t hi = font.kerningCount();
183 while (lo < hi) {
184 const size_t mid = lo + (hi - lo) / 2;
185 const size_t o = kerningEntryOffset(font, mid);
186 const uint32_t l = font.u32(o);
187 const uint32_t r = font.u32(o + 4);
188 if (l < left || (l == left && r < right)) {
189 lo = mid + 1;
190 } else {
191 hi = mid;
192 }
193 }
194 if (lo >= font.kerningCount()) return 0;
195 const size_t o = kerningEntryOffset(font, lo);
196 if (font.u32(o) == left && font.u32(o + 4) == right) return font.i16(o + 8);
197 return 0;
198}
