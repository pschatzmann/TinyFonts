58 // Draws a single line (no wrapping, '\n' is not special). Returns the
59 // advance width of the line.
60 int drawLine(int x, int y, const char* begin, const char* end) {
61 const int baseline = y + _font.ascent();
62 int pen = x;
63 uint32_t prev = 0;
64 bool hasPrev = false;
65 const char* p = begin;
66 while (p < end) {
67 const uint32_t cp = decodeUtf8(p, end);
68 if (cp == '\n') break;
69 if (hasPrev) pen += kerning(_font, prev, cp);
70 Glyph g;
71 if (findGlyph(_font, cp, g)) {
72 renderGlyph(g, pen, baseline);
73 pen += g.advance;
74 }
75 prev = cp;
76 hasPrev = true;
77 }
78 return pen - x;
79 }

81 // Draws text, honouring '\n' and wrapping at maxWidth (<= 0: no wrap).
82 // Returns the size of the drawn block.
83 TextBox drawText(int x, int y, const char* begin, const char* end,
84 int maxWidth = 0) {
85 const int lineHeight = _font.lineHeight();
86 TextBox box;
87 int line = 0;
88 box.lines = wrapText(_font, begin, end, maxWidth,
89 [&](const char* b, const char* e, int w) {
90 drawLine(x, y + line * lineHeight, b, e);
91 if (w > box.width) box.width = w;
92 ++line;
93 });
94 box.height = box.lines * lineHeight;
95 return box;
96 }

98 // Convenience overload for a NUL-terminated string.
99 TextBox drawText(int x, int y, const char* text, int maxWidth = 0) {
100 const char* end = text;
101 while (*end) ++end;
102 return drawText(x, y, text, end, maxWidth);
103 }

105private:
106 void renderGlyph(const Glyph& g, int penX, int baseline) {
107 if (g.width == 0 || g.height == 0) return;
108
109 const uint8_t bpp = _font.bitsPerPixel();
110 const uint32_t maxValue = (1u << bpp) - 1u;
111 const size_t stride = (static_cast<size_t>(g.width) * bpp + 7) / 8;
112 const size_t base = _font.bitmapBase() + g.bitmapOffset;
113 const int x0 = penX + g.left;
114 const int y0 = baseline - g.top;
115
116 for (int row = 0; row < g.height; ++row) {
117 const int y = y0 + row;
118 if (_clip && (y < _cy0 || y >= _cy1)) continue;
119 const size_t rowBase = base + static_cast<size_t>(row) * stride;
120
121 for (int col = 0; col < g.width; ++col) {
122 const int x = x0 + col;
123 if (_clip && (x < _cx0 || x >= _cx1)) continue;
124
125 const size_t bit = static_cast<size_t>(col) * bpp;
126 const uint8_t byte = _font.u8(rowBase + bit / 8);
127 const uint8_t shift = static_cast<uint8_t>(8 - bpp - (bit % 8));
128 const uint32_t value = (byte >> shift) & maxValue;
129 if (value == 0) continue;
130
131 const uint8_t coverage =
132 static_cast<uint8_t>((value * 255u + maxValue / 2) / maxValue);
133 _display.drawPixel(x, y, coverage);
134 }
135 }
136 }

139 FontDisplay& _display;
140 bool _clip = false;
141 int _cx0 = 0, _cy0 = 0, _cx1 = 0, _cy1 = 0;
