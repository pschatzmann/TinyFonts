122 void renderGlyph(const Glyph& g, int penX, int baseline) {
123 if (g.width == 0 || g.height == 0) return;
124
125 const uint8_t bpp = _font.bitsPerPixel();
126 const uint32_t maxValue = (1u << bpp) - 1u;
127 const size_t stride = (static_cast<size_t>(g.width) * bpp + 7) / 8;
128 const size_t base = _font.bitmapBase() + g.bitmapOffset;
129 const int x0 = penX + g.left;
130 const int y0 = baseline - g.top;
131
132 for (int row = 0; row < g.height; ++row) {
133 const int y = y0 + row;
134 if (_clip && (y < _cy0 || y >= _cy1)) continue;
135 const size_t rowBase = base + static_cast<size_t>(row) * stride;
136
137 for (int col = 0; col < g.width; ++col) {
138 const int x = x0 + col;
139 if (_clip && (x < _cx0 || x >= _cx1)) continue;
140
141 const size_t bit = static_cast<size_t>(col) * bpp;
142 const uint8_t byte = _font.u8(rowBase + bit / 8);
143 const uint8_t shift = static_cast<uint8_t>(8 - bpp - (bit % 8));
144 const uint32_t value = (byte >> shift) & maxValue;
145 if (value == 0) continue;
146
147 const uint8_t coverage =
148 static_cast<uint8_t>((value * 255u + maxValue / 2) / maxValue);
149 _display.drawPixel(x, y, coverage);
150 }
151 }
152 }
